async function loadPartial(id, file) { const el = document.getElementById(id); if (!el) return; try { const res = await fetch(file, { cache: "no-store" }); if (!res.ok) throw new Error("Fetch failed"); el.innerHTML = await res.text(); } catch (e) { console.error("Partial load error:", file, e); } } document.addEventListener("DOMContentLoaded", () => { loadPartial("site-header", "/partials/header.html"); loadPartial("site-footer", "/partials/footer.html"); });
